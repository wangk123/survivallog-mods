**Goal:** 落地 docs/07 设计的游戏资料库：版本化快照目录 + 两个对比工具转正 + ConfigDumpProbe 表探针 + moddeps 依赖清单与兼容核对 + 一键同步脚本。

- [ ] **Step 2: 写 sigscan.js（重构版：模块 + CLI）**

`tools/gamekb/sigscan.js` 全文：

```js
// sigscan.js — gamekb 工具：在 GameAssembly.dll 文件内静态扫描字节签名（PE 段表自解析）
// CLI: node sigscan.js <GameAssembly.dll>   （内置 FridgeChill TalentCapBytePatch 签名，输出 RESULT 行）
// 模块导出（modcompat_check.js 复用）:
//   parsePE(buf) -> { sections:[{name,virtSize,virtAddr,rawSize,rawPtr}], il2cpp }
//   rva2Off(sections, rva) -> 文件偏移（-1=不在任何段）
//   hexToSig("F30F5CD9????????C3") -> number[]（?? -> -1 通配）
//   scanPattern(secBuf, sig) -> number[]（段缓冲内全部命中偏移，升序）
//   readRel32Target(peBuf, sections, il2cpp, secBuf, hitOff, rel32ByteOff) -> { insnRva, slotRva }
'use strict';
const fs = require('fs');

function parsePE(pe) {
  const u16 = o => pe.readUInt16LE(o), u32 = o => pe.readUInt32LE(o);
  const peOff = u32(0x3c), nSec = u16(peOff + 6), optSize = u16(peOff + 20);
  const sections = [];
  for (let i = 0; i < nSec; i++) {
    const b = peOff + 24 + optSize + i * 40;
    sections.push({
      name: pe.toString('latin1', b, b + 8).replace(/\0+$/, ''),
      virtSize: u32(b + 8), virtAddr: u32(b + 12), rawSize: u32(b + 16), rawPtr: u32(b + 20),
    });
  }
  return { sections, il2cpp: sections.find(s => s.name === 'il2cpp') };
}
function rva2Off(sections, rva) {
  for (const s of sections)
    if (rva >= s.virtAddr && rva < s.virtAddr + Math.max(s.virtSize, s.rawSize)) return s.rawPtr + (rva - s.virtAddr);
  return -1;
}
function hexToSig(sigHex) {
  const out = [];
  for (let i = 0; i < sigHex.length; i += 2) {
    const h = sigHex.substr(i, 2);
    out.push(h === '??' ? -1 : parseInt(h, 16));
  }
  return out;
}
function scanPattern(buf, sig) {
  const hits = [];
  for (let i = 0; i <= buf.length - sig.length; i++) {
    let ok = true;
    for (let j = 0; j < sig.length; j++) { if (sig[j] < 0) continue; if (buf[i + j] !== sig[j]) { ok = false; break; } }
    if (ok) hits.push(i);
  }
  return hits;
}
// hitOff：命中在 il2cpp 段缓冲内的偏移；rel32ByteOff：rel32 在签名内的字节下标
function readRel32Target(peBuf, sections, il2cpp, secBuf, hitOff, rel32ByteOff) {
  const insnRva = il2cpp.virtAddr + hitOff;
  const rel32 = secBuf.readInt32LE(hitOff + rel32ByteOff);
  const slotRva = insnRva + rel32ByteOff + 4 + rel32;
  return { insnRva, slotRva };
}

module.exports = { parsePE, rva2Off, hexToSig, scanPattern, readRel32Target };

// ---- CLI ----
if (require.main === module) {
  const pe = fs.readFileSync(process.argv[2]);
  const { sections, il2cpp } = parsePE(pe);
  if (!il2cpp) { console.log('NO il2cpp section'); process.exit(1); }
  console.log(`il2cpp section: VA=0x${il2cpp.virtAddr.toString(16)} size=0x${il2cpp.virtSize.toString(16)} raw=0x${il2cpp.rawPtr.toString(16)}`);
  // F30F5CD9 F30F5FD3 F30F59D0 F30F1005 <rel32 通配> F30F5FC2 C3 —— rel32 在第 16 字节
  const sig = hexToSig('F30F5CD9F30F5FD3F30F59D0F30F1005????????F30F5FC2C3');
  const buf = pe.subarray(il2cpp.rawPtr, il2cpp.rawPtr + il2cpp.rawSize);
  const hits = scanPattern(buf, sig);
  hits.forEach((h, n) => {
    const { insnRva, slotRva } = readRel32Target(pe, sections, il2cpp, buf, h, 16);
    const off = rva2Off(sections, slotRva);
    const val = off >= 0 ? pe.readFloatLE(off) : NaN;
    console.log(`HIT#${n + 1} insnRVA=0x${insnRva.toString(16)} -> slotRVA=0x${slotRva.toString(16)} value=${val}`);
  });
  console.log(hits.length === 0 ? 'RESULT: SIGNATURE NOT FOUND' : `RESULT: ${hits.length} hit(s)`);
}
```

using System.Collections.Generic;
using System.Globalization;
using System.Text;
using System.Text.Json;
using BepInEx;
using BepInEx.Logging;
using BepInEx.Unity.IL2CPP;

using Il2CppInterop.Runtime.Injection;
using UnityEngine;
using Object = UnityEngine.Object;

namespace SurvivalLog.ConfigDumpProbe;

/// <summary>
/// gamekb 表数据探针 v1（一次性取证，绝不入发布套件；设计见 docs/07 §6）。
/// 主菜单阶段轮询 ConfigManager：125 个 _Config_*_Dict 全部就绪且两轮指纹相同后，
/// 全表序列化为 JSON 落 BepInEx\gamekb_tables\&lt;RuntimeAssets版本&gt;\，以 _DONE.txt 自证。
/// 红线：不写 cfg、不挂 Harmony、不用 il2cpp 泛型方法（自写泛型经 MakeGenericMethod 调用）。
/// </summary>
[BepInPlugin(Guid, Name, Version)]
public sealed class ConfigDumpProbePlugin : BasePlugin
{
    public const string Guid = "com.local.survivallog.configdumpprobe";
    public const string Name = "ConfigDumpProbe";
    public const string Version = "1.0.0";

    internal static new ManualLogSource Log;

    public override void Load()
    {
        Log = base.Log;
        if (!ClassInjector.IsTypeRegisteredInIl2Cpp<ProbeTicker>())
            ClassInjector.RegisterTypeInIl2Cpp<ProbeTicker>();
        var go = new GameObject("ConfigDumpProbe.Ticker");
        Object.DontDestroyOnLoad(go);
        go.AddComponent<ProbeTicker>();
        Log.LogInfo("[ConfigDumpProbe] v1 loaded（全量 Config 表 dump -> gamekb_tables/<资源版本>）");

}

internal sealed class ProbeTicker : MonoBehaviour
{
    public ProbeTicker(IntPtr ptr) : base(ptr) { }
    public ProbeTicker() { }

    private float _poll;
    private string _lastFp = "";
    private int _stableRounds;
    private bool _done;

    // (属性名去掉 _Config_ 前缀和 _Dict 后缀, 属性反射, 行类型)
    private readonly List<(string name, System.Reflection.PropertyInfo pi, Type rowType)> _dicts = new();
    // 降级字段（列名）汇总，随 _index.json 落盘
    private readonly SortedSet<string> _degraded = new();

    private static GameCore.HotUpdate.ConfigManager GetCm()
    {
        try
        {
            if (!GameCore.HotUpdate.BaseSingleton<GameCore.HotUpdate.ConfigManager>.IsInstanceCreated)
                return null;
            return GameCore.HotUpdate.BaseSingleton<GameCore.HotUpdate.ConfigManager>.Instance;
        }
        catch { return null; }
    }

    private static string RuntimeAssetsVersion()
    {
        try
        {
            var p = System.IO.Path.Combine(Application.dataPath,
                "StreamingAssets", "PackageManifest", "MainPackage", "RuntimeAssets_MainPackage.version");
            return System.IO.File.ReadAllText(p).Trim();
        }
        catch { return "unknown"; }
    }

    public void Update()
    {
        if (_done) return;
        _poll += Time.deltaTime;
        if (_poll < 2f) return;
        _poll = 0f;

        var cm = GetCm();
        if (cm == null) return;

        if (_dicts.Count < 100)
        {
            // 发现/再发现：表可能分批注册，凑满 100 个前每轮清空重枚举（interop 程序集普通反射）
            _dicts.Clear();
            foreach (var pi in typeof(GameCore.HotUpdate.ConfigManager).GetProperties())
            {
                var m = System.Text.RegularExpressions.Regex.Match(pi.Name, "^_Config_(.+)_Dict$");
                if (!m.Success) continue;
                var args = pi.PropertyType.GetGenericArguments();
                if (args.Length != 2) continue;
                _dicts.Add((m.Groups[1].Value, pi, args[1]));
            }
            if (_dicts.Count < 100) return; // 表未全量注册，下轮再看
            ActionProbeLog($"发现 {_dicts.Count} 个 _Config_*_Dict 属性");
        }

        // 指纹：非空表数:总行数。两轮相同（且非空表>=100）才算加载完毕
        int tables = 0, rows = 0;
        foreach (var (_, pi, _) in _dicts)
        {
            try
            {
                var d = pi.GetValue(cm, null);
                if (d == null) continue;
                var c = (int)d.GetType().GetProperty("Count").GetValue(d, null);
                if (c > 0) { tables++; rows += c; }
            }
            catch { }
        }
        var fp = $"{tables}:{rows}";
        if (fp == _lastFp && tables >= 100) _stableRounds++;
        else _stableRounds = 0;
        _lastFp = fp;
        if (_stableRounds < 2) return;

        _done = true;
        try { DumpAll(cm); }
        catch (Exception e) { ActionProbeLog("dump 失败: " + e); }
    }

    private static void ActionProbeLog(string s) => ConfigDumpProbePlugin.Log.LogInfo("[ConfigDumpProbe] " + s);

    // ---- dump ----
    private static readonly System.Reflection.BindingFlags BI =
        System.Reflection.BindingFlags.Public | System.Reflection.BindingFlags.Instance;

    private void DumpAll(GameCore.HotUpdate.ConfigManager cm)
    {
        var ver = RuntimeAssetsVersion();
        var dir = System.IO.Path.Combine(Paths.BepInExRootPath, "gamekb_tables", ver);
        System.IO.Directory.CreateDirectory(dir);
        ActionProbeLog($"开始 dump：ver={ver} 目标={dir}");

        var index = new List<object>();
        long totalRows = 0;
        foreach (var (name, pi, rowType) in _dicts)
        {
            object dictObj = null;
            try { dictObj = pi.GetValue(cm, null); } catch { }
            int count = 0;
            if (dictObj != null)
            {
                try { count = (int)dictObj.GetType().GetProperty("Count").GetValue(dictObj, null); } catch { }
            }
            var rowsJson = new List<Dictionary<string, object>>();
            if (dictObj != null && count > 0)
            {
                // 自写泛型方法 + MakeGenericMethod（普通 CLR 反射，绕开 interop 泛型限制）
                // 单表失败（key 非整型/泛型约束不符等）只降级该表，不中断全量
                try
                {
                    var mi = typeof(ProbeTicker).GetMethod(nameof(EnumerateRows),
                        System.Reflection.BindingFlags.NonPublic | System.Reflection.BindingFlags.Static)
                        .MakeGenericMethod(typeof(int), rowType);
                    mi.Invoke(null, new object[] { dictObj, rowsJson, _degraded, name });
                }
                catch (Exception e)
                {
                    _degraded.Add($"{name}:<enumerate-failed:{(e.InnerException ?? e).GetType().Name}>");
                }
            }
            totalRows += count;
            var columns = new List<string>();
            if (rowsJson.Count > 0) foreach (var k in rowsJson[0].Keys) columns.Add(k);
            var file = System.IO.Path.Combine(dir, $"Config_{name}.json");
            System.IO.File.WriteAllText(file, JsonSerializer.Serialize(new
            { table = $"Config_{name}", rows = count, columns, data = rowsJson }),
                new UTF8Encoding(false));
            index.Add(new { name = $"Config_{name}", rows = count, columns,
                sha256 = Sha256File(file), degraded = new string[0] });
            ActionProbeLog($"  Config_{name}: {count} 行");
        }

        var idxPath = System.IO.Path.Combine(dir, "_index.json");
        System.IO.File.WriteAllText(idxPath, JsonSerializer.Serialize(new
        { runtime_assets_version = ver, generated_at = DateTime.Now.ToString("o"),
          total_rows = totalRows, degraded_columns = _degraded, tables = index }),
            new UTF8Encoding(false));
        System.IO.File.WriteAllText(System.IO.Path.Combine(dir, "_DONE.txt"),
            $"tables={_dicts.Count} rows={totalRows} deg={_degraded.Count} ver={ver} at={DateTime.Now:yyyy-MM-dd HH:mm:ss}\n",
            new UTF8Encoding(false));
        ActionProbeLog($"DONE：tables={_dicts.Count} rows={totalRows} deg={_degraded.Count}");
    }

    private static string Sha256File(string path)
    {
        using var sha = System.Security.Cryptography.SHA256.Create();
        using var f = System.IO.File.OpenRead(path);
        return BitConverter.ToString(sha.ComputeHash(f)).Replace("-", "").ToLowerInvariant();
    }

    // 闭式泛型：TRow=interop 行类型；调用方经 MakeGenericMethod 进入
    private static void EnumerateRows<TKey, TRow>(object dictObj, List<Dictionary<string, object>> outRows,
        SortedSet<string> degraded, string tableName) where TRow : Il2CppObjectBase
    {
        var dict = dictObj as Il2CppSystem.Collections.Generic.Dictionary<TKey, TRow>;
        if (dict == null) { degraded.Add($"{tableName}:<dict-cast-failed>"); return; }
        foreach (var kv in dict)
        {
            var row = kv.Value;
            if (row == null) continue;
            var obj = new Dictionary<string, object>();
            foreach (var p in typeof(TRow).GetProperties(BI))
            {
                object v;
                try { v = p.GetValue(row, null); } catch { continue; }
                if (v == null) { obj[p.Name] = null; continue; }
                obj[p.Name] = ValueToJson(v, degraded, $"{tableName}.{p.Name}", 0);
            }
            outRows.Add(obj);
        }
    }

    private static object ValueToJson(object v, SortedSet<string> degraded, string colId, int depth)
    {
        var t = v.GetType();
        if (v is string || t.IsPrimitive || t.IsEnum) return v;
        // List 代理：Il2CppSystem.Collections.Generic.List`1，用 Count+Item 反射枚举
        if (depth < 3 && t.Namespace == "Il2CppSystem.Collections.Generic" && t.Name.StartsWith("List`1"))
        {
            try
            {
                var cnt = (int)t.GetProperty("Count").GetValue(v, null);
                var item = t.GetProperty("Item");
                var arr = new List<object>();
                for (int i = 0; i < cnt; i++)
                    arr.Add(ValueToJson(item.GetValue(v, new object[] { i }), degraded, colId, depth + 1));
                return arr;
            }
            catch (Exception e) { degraded.Add($"{colId}:{e.GetType().Name}"); }
        }
        // 数组代理（Il2CppArrayBase 派生）：尽力按 IEnumerable
        if (depth < 3 && v is System.Collections.IEnumerable en)
        {
            try
            {
                var arr = new List<object>();
                foreach (var x in en) arr.Add(ValueToJson(x, degraded, colId, depth + 1));
                return arr;
            }
            catch (Exception e) { degraded.Add($"{colId}:{e.GetType().Name}"); }
        }
        // 其余（嵌套配置对象/字典等）：降级 ToString 并标注（设计 §6 允许）
        if (v is Il2CppObjectBase) degraded.Add(colId);
        return "$deg:" + t.Name + ":" + Convert.ToString(v, CultureInfo.InvariantCulture);
    }
}

`tools/gamekb/modcompat_check.js` 全文：

```js
// modcompat_check.js — gamekb 工具：按 moddeps.json 核对某版 gamekb 是否满足全套件依赖
// CLI: node modcompat_check.js <gamekb版本目录> [--moddeps <path>]（默认同目录 moddeps.json）
// 判定：类型/方法/访问器/行类/字段缺失或签名命中异常 = RED；参数个数不符 = WARN（运行时兜底）
// 表 schema（tables/_index.json 存在时）：字段不在 columns = RED（表结构变更）
// exit: 0=全绿 1=有红 2=用法/IO 错误
'use strict';
const fs = require('fs');
const path = require('path');
const { parseDump } = require('./dumpdiff.js');
const { parsePE, rva2Off, hexToSig, scanPattern, readRel32Target } = require('./sigscan.js');

const a = process.argv.slice(2).filter(x => !x.startsWith('--'));
const verDir = a[0];
const mdPath = process.argv.includes('--moddeps')
  ? process.argv[process.argv.indexOf('--moddeps') + 1]
  : path.join(__dirname, 'moddeps.json');
if (!verDir || !fs.existsSync(verDir)) { console.error('usage: node modcompat_check.js <gamekb版本目录> [--moddeps <path>]'); process.exit(2); }

const deps = JSON.parse(fs.readFileSync(mdPath, 'utf8'));
const dumpPath = path.join(verDir, 'il2cpp', 'dump.cs');
if (!fs.existsSync(dumpPath)) { console.error('no dump.cs under ' + dumpPath); process.exit(2); }
console.log(`# modcompat @ ${path.basename(verDir)}`);
const types = parseDump(dumpPath);

// 方法行匹配：<Method>（ 后面数参数（顶层逗号+1，粗对）
function methodHits(body, method) {
  const re = new RegExp(`[ .]${method}\\(`, 'g');
  const out = [];
  for (const l of body) { re.lastIndex = 0; if (re.test(l)) out.push(l); }
  return out;
}
function paramCount(line) {
  const i = line.indexOf('('); if (i < 0) return -1;
  let d = 0, c = 0;
  for (let j = i + 1; j < line.length; j++) {
    const ch = line[j];
    if (ch === '(' || ch === '<') d++;
    else if (ch === ')' || ch === '>') { if (d === 0) return c; d--; }
    else if (ch === ',' && d === 0) c++;
  }
  return c + 1;
}
// 字段/属性行匹配：词边界 + 行尾 ; 或 { }
function fieldHit(body, field) {
  const re = new RegExp(`[ .]${field}\\s*[;{]`);
  return body.some(l => re.test(l));
}

let red = 0, warn = 0;
const R = m => { red++; console.log(`  [RED]   ${m}`); };
const W = m => { warn++; console.log(`  [WARN]  ${m}`); };
const G = m => console.log(`  [ok]    ${m}`);

// tables/_index.json（可选）：行类 -> columns
const idxPath = path.join(verDir, 'tables', '_index.json');
const tableCols = new Map();
let tablesLoaded = false;
if (fs.existsSync(idxPath)) {
  tablesLoaded = true;
  const idx = JSON.parse(fs.readFileSync(idxPath, 'utf8'));
  for (const t of idx.tables || []) tableCols.set(t.name, t.columns || []);
}

const cfgMgr = types.get('GameCore.HotUpdate.ConfigManager');

for (const [mod, dep] of Object.entries(deps)) {
  console.log(`## ${mod}`);
  for (const h of dep.harmony || []) {
    const t = types.get(h.type);
    if (!t) { R(`${h.type} 类型缺失`); continue; }
    const hits = methodHits(t.body, h.method);
    if (hits.length === 0) { R(`${h.type}.${h.method} 方法缺失`); continue; }
    if (h.params != null) {
      const ok = hits.some(l => paramCount(l) === h.params);
      ok ? G(`${h.type}.${h.method}(${h.params}参)`) : W(`${h.type}.${h.method} 参数个数不符：期望 ${h.params}，实际 ${hits.map(paramCount).join('/')}`);
    } else G(`${h.type}.${h.method}`);
  }
  for (const c of dep.configFields || []) {
    if (!cfgMgr) { R('ConfigManager 类型缺失'); continue; }
    if (!cfgMgr.body.some(l => l.includes(c.accessor))) { R(`ConfigManager.${c.accessor} 访问器缺失`); continue; }
    const row = types.get('GameCore.HotUpdate.' + c.rowType);
    if (!row) { R(`${c.rowType} 行类缺失`); continue; }
    G(`${c.accessor} ✓`);
    for (const f of c.fields) {
      if (!fieldHit(row.body, f)) { R(`${c.rowType}.${f} 字段缺失`); continue; }
      if (tablesLoaded) {
        const cols = tableCols.get(c.rowType);
        if (cols && cols.length && !cols.includes(f)) { R(`${c.rowType}.${f} 不在 tables/_index.json columns（表结构变更）`); continue; }
      }
      G(`${c.rowType}.${f}`);
    }
  }
  for (const s of dep.nativeSigs || []) {
    const pePath = path.join(verDir, 'binaries', 'GameAssembly.dll');
    if (!fs.existsSync(pePath)) { W(`${s.name}: binaries/GameAssembly.dll 缺失（partial 基线），跳过`); continue; }
    const pe = fs.readFileSync(pePath);
    const { sections, il2cpp } = parsePE(pe);
    const buf = pe.subarray(il2cpp.rawPtr, il2cpp.rawPtr + il2cpp.rawSize);
    const hits = scanPattern(buf, hexToSig(s.sigHex));
    if (hits.length === 0) { R(`${s.name} 签名 0 命中（游戏实现已变，运行时会安全关闭）`); continue; }
    if (s.expectHits != null && hits.length !== s.expectHits) W(`${s.name} 命中 ${hits.length} 处（期望 ${s.expectHits}，运行时取第一处，请人工确认语义）`);
    if (s.expectOldSlot != null) {
      const { slotRva } = readRel32Target(pe, sections, il2cpp, buf, hits[0], s.rel32ByteOff);
      const off = rva2Off(sections, slotRva);
      const val = off >= 0 ? pe.readFloatLE(off) : NaN;
      if (Math.abs(val - s.expectOldSlot) > 1e-4) R(`${s.name} 旧槽值 ${val} ≠ ${s.expectOldSlot}`); else G(`${s.name} 签名 ${hits.length} 命中，旧槽 ${val}`);
    } else G(`${s.name} 签名 ${hits.length} 命中`);
  }
}

console.log(`\n== 汇总：RED=${red} WARN=${warn} ${red === 0 ? '(ALL GREEN)' : '(HAS RED)'} ==`);
process.exit(red === 0 ? 0 : 1);
```

`tools/gamekb/gamekb_sync.ps1` 全文（ASCII only——注释也用英文，中文说明去哪了看 docs/07）：

```powershell
# gamekb_sync.ps1 - one-command gamekb sync after a game update (see docs/07 section 5)
# Usage: powershell -NoProfile -ExecutionPolicy Bypass -File tools\gamekb\gamekb_sync.ps1 [-AutoProbe]
#   -AutoProbe : also launch the game via steam:// to run ConfigDumpProbe, wait for _DONE.txt,
#                then stop the game and collect tables. Aborts if the game is already running.
param([switch]$AutoProbe)

$ErrorActionPreference = 'Stop'
$repo = Split-Path -Parent (Split-Path -Parent $MyInvocation.MyCommand.Path)   # repo root
$game = $null

# ---- 1. locate game dir: registry -> libraryfolders.vdf -> appmanifest ----
function Find-GameDir {
    $steamPath = (Get-ItemProperty 'HKCU:\SOFTWARE\Valve\Steam' -ErrorAction SilentlyContinue).SteamPath
    if (-not $steamPath) { $steamPath = 'D:\APP\Steam' }
    $lf = Join-Path $steamPath 'steamapps\libraryfolders.vdf'
    if (-not (Test-Path $lf)) { throw "libraryfolders.vdf not found: $lf" }
    $libs = @((Get-Content $lf | Select-String '"path"\s+"([^"]+)"').Matches | ForEach-Object { $_.Groups[1].Value })
    foreach ($lib in $libs) {
        $mani = Join-Path $lib 'steamapps\appmanifest_4164790.acf'
        if (Test-Path $mani) {
            $dir = Join-Path $lib ('steamapps\common\' + ((Get-Content $mani | Select-String '"installdir"\s+"([^"]+)"').Matches[0].Groups[1].Value))
            if (Test-Path (Join-Path $dir 'GameAssembly.dll')) { return ,$dir }
        }
    }
    throw 'Survival Log install dir not found in any Steam library'
}
$game = Find-GameDir
Write-Host ("game dir: " + $game)

# ---- 2. read buildid + runtime assets version; idempotence gate ----
$mani4164790 = Get-ChildItem (Split-Path -Parent (Split-Path -Parent $game)) -Recurse -Filter 'appmanifest_4164790.acf' -ErrorAction SilentlyContinue | Select-Object -First 1
$buildId = [int]((Get-Content $mani4164790.FullName | Select-String '"buildid"\s+"(\d+)"').Matches[0].Groups[1].Value)
$raVer = (Get-Content (Join-Path $game 'SurvivalLog_Data\StreamingAssets\PackageManifest\MainPackage\RuntimeAssets_MainPackage.version') -ErrorAction SilentlyContinue)
if (-not $raVer) { $raVer = 'unknown' }
$kbRoot = Join-Path $repo 'gamekb'
$existing = Get-ChildItem $kbRoot -Directory -ErrorAction SilentlyContinue | Where-Object { $_.Name -match ('_' + $buildId + '$') }
if ($existing) { Write-Host ("no update: buildid {0} already captured as {1}" -f $buildId, $existing[0].Name); exit 0 }

# ---- 3. pre-flight: dumper exe present + no unclaimed outputs (BEFORE creating dirs) ----
$dumperDir = Join-Path $repo 'tools\il2cppdumper'
$dumperExe = Join-Path $dumperDir 'Il2CppDumper.exe'
if (-not (Test-Path $dumperExe)) {
    Write-Host ('ERROR: Il2CppDumper.exe missing under ' + $dumperDir + ' (gitignored; fetch Perfare/Il2CppDumper v6.7.46 net6-win and put exe+config.json+runtimeconfig there)')
    exit 1
}
$leftover = Get-ChildItem $dumperDir -File | Where-Object { $_.Name -in 'dump.cs','script.json','stringliteral.json','il2cpp.h' }
if ($leftover) { Write-Host ('ERROR: unclaimed dumper outputs in exe dir: ' + ($leftover.Name -join ',')); exit 1 }

# ---- 4. create version dir, copy binaries, meta ----
$stamp = Get-Date -Format 'yyyy-MM-dd'
$verDir = Join-Path $kbRoot ("{0}_{1}" -f $stamp, $buildId)
New-Item -ItemType Directory -Force -Path (Join-Path $verDir 'il2cpp'), (Join-Path $verDir 'binaries') | Out-Null
Copy-Item (Join-Path $game 'GameAssembly.dll') (Join-Path $verDir 'binaries\') -Force
Copy-Item (Join-Path $game 'SurvivalLog_Data\il2cpp_data\Metadata\global-metadata.dat') (Join-Path $verDir 'binaries\') -Force
$sha1 = (Get-FileHash (Join-Path $verDir 'binaries\GameAssembly.dll') -Algorithm SHA256).Hash
$sha2 = (Get-FileHash (Join-Path $verDir 'binaries\global-metadata.dat') -Algorithm SHA256).Hash
$meta = [ordered]@{
    buildid = $buildId; runtime_assets_version = $raVer
    unity_version = ''; captured_at = (Get-Date).ToString('o')
    sha256 = [ordered]@{ 'GameAssembly.dll' = $sha1; 'global-metadata.dat' = $sha2 }
    tools = [ordered]@{ il2cppdumper = '6.7.46'; dumpdiff = '1.0' }
    partial = $false; notes = ''
}
($meta | ConvertTo-Json -Depth 4) | Out-File (Join-Path $verDir 'meta.json') -Encoding ascii
Write-Host ("capturing " + $verDir)

# ---- 5. run Il2CppDumper (outputs land in exe dir - move, never copy) ----
$proc = Start-Process -FilePath $dumperExe -ArgumentList ('"' + (Join-Path $game 'GameAssembly.dll') + '" "' + (Join-Path $game 'SurvivalLog_Data\il2cpp_data\Metadata\global-metadata.dat') + '" "' + $dumperDir + '"') -Wait -PassThru -WindowStyle Hidden
Write-Host ("dumper exit code: " + $proc.ExitCode)
foreach ($f in 'dump.cs','script.json','stringliteral.json') {
    $src = Join-Path $dumperDir $f
    if (Test-Path $src) { Move-Item $src (Join-Path $verDir ('il2cpp\' + $f)) -Force }
}
Remove-Item (Join-Path $dumperDir 'il2cpp.h') -ErrorAction SilentlyContinue
Remove-Item (Join-Path $dumperDir 'DummyDll') -Recurse -ErrorAction SilentlyContinue
if (-not (Test-Path (Join-Path $verDir 'il2cpp\dump.cs'))) { Write-Host 'ERROR: dumper produced no dump.cs'; exit 1 }

# ---- 6. webui mirror ----
robocopy (Join-Path $game 'SurvivalLog_Data\StreamingAssets\WebUI') (Join-Path $verDir 'webui') /MIR /NJH /NJS /NP | Out-Null
if ($LASTEXITCODE -ge 8) { Write-Host ('ERROR: robocopy failed ' + $LASTEXITCODE); exit 1 }

# ---- 7. diff vs previous version ----
$prev = Get-ChildItem $kbRoot -Directory | Where-Object { $_.Name -ne (Split-Path -Leaf $verDir) } | Sort-Object Name -Descending | Select-Object -First 1
$node = Get-Command node -ErrorAction SilentlyContinue
if ($prev -and $node) {
    $diffMd = Join-Path $verDir ('diff_from_' + ($prev.Name -replace '_\d+$','') + '.md')
    & node (Join-Path $repo 'tools\gamekb\dumpdiff.js') (Join-Path $prev.FullName 'il2cpp\dump.cs') (Join-Path $verDir 'il2cpp\dump.cs') ('--moddeps ' + (Join-Path $repo 'tools\gamekb\moddeps.json')) | Out-File $diffMd -Encoding utf8
    & node (Join-Path $repo 'tools\gamekb\modcompat_check.js') $verDir | Tee-Object -FilePath $diffMd -Append
    Write-Host ("diff written: " + $diffMd)
}

# ---- 8. tables: manual or -AutoProbe ----
$doneMarker = Join-Path $game ('BepInEx\gamekb_tables\' + $raVer + '\_DONE.txt')
if ($AutoProbe) {
    $running = Get-Process SurvivalLog -ErrorAction SilentlyContinue
    if ($running) { Write-Host 'ABORT -AutoProbe: game is running (author may be playing). Rerun when idle.'; exit 1 }
    if (-not (Test-Path (Join-Path $game 'BepInEx\plugins\SurvivalLog.ConfigDumpProbe.dll'))) {
        Write-Host 'ERROR: ConfigDumpProbe.dll not deployed to BepInEx\plugins (build SurvivalLog.ConfigDumpProbe and copy first)'; exit 1
    }
    Start-Process 'steam://rungameid/4164790'
    $deadline = (Get-Date).AddMinutes(5)
    while (-not (Test-Path $doneMarker) -and (Get-Date) -lt $deadline) { Start-Sleep -Seconds 5 }
    if (-not (Test-Path $doneMarker)) { Write-Host 'ERROR: _DONE.txt not appeared in 5min (check probe deploy / game start)'; exit 1 }
    Get-Process SurvivalLog -ErrorAction SilentlyContinue | Stop-Process -Force
    Start-Sleep -Seconds 3
    $still = Get-Process SurvivalLog -ErrorAction SilentlyContinue
    if ($still) { Write-Host 'WARN: SurvivalLog still alive after stop; tables not collected automatically'; exit 1 }
} 
if (Test-Path $doneMarker) {
    $tablesSrc = Join-Path $game ('BepInEx\gamekb_tables\' + $raVer)
    $tablesDst = Join-Path $verDir 'tables'
    if (Test-Path $tablesDst) { Remove-Item $tablesDst -Recurse -Force }
    Move-Item $tablesSrc $tablesDst
    Write-Host ("tables collected: " + (Get-Content $doneMarker -TotalCount 1))
} else {
    Write-Host ('TABLES PENDING: launch the game once (probe writes BepInEx\gamekb_tables\' + $raVer + '), then rerun this script to collect.')
}

Write-Host ('DONE buildid=' + $buildId + ' raVer=' + $raVer)
Write-Host 'manual step: fill meta.json notes + review diff_from_*.md'
exit 0
```
